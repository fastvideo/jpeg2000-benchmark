# CMakeLists.txt for opj_bench
# version 03, 14 September 2026
#
# Builds against a tree that OpenJPEG was BUILT in, not installed from, which
# is the case on the measurement machine:
#
#   D:/_Test/openjpeg-master            <- OPJ_ROOT, the source tree
#   D:/_Test/openjpeg-master/bin        <- OPJ_BUILD, the CMake build tree
#
# In a build tree the two headers live apart: openjpeg.h stays in the source
# at src/lib/openjp2, while opj_config.h is generated into the build tree at
# src/lib/openjp2. openjpeg.h needs both, so both directories go on the
# include path. An installed OpenJPEG (or a Linux package) is also found.
#
#   cmake -S . -B build -DOPJ_ROOT=D:/_Test/openjpeg-master
#   cmake --build build --config Release
#
# ---------------------------------------------------------------------------
# TWO BUILDS, AND WHY THE SECOND ONE IS ABOUT THE LIBRARY
# ---------------------------------------------------------------------------
#
# The article says the numbers were taken from an ordinary Release build,
# without vector instructions. To answer "and what if it had them", a second
# pair is built - and the flag has to go into OPENJPEG, not into this file.
# This file does almost no arithmetic; all the work is inside the library.
#
# Step 1. Build OpenJPEG twice, into two separate trees. OpenJPEG is C, so the
# flag goes into the C flags, not the C++ ones.
#
#   cmake -S <openjpeg> -B <openjpeg>/bin
#   cmake --build <openjpeg>/bin --config Release
#
#   cmake -S <openjpeg> -B <openjpeg>/bin-avx2 -DCMAKE_C_FLAGS="/arch:AVX2"
#   cmake --build <openjpeg>/bin-avx2 --config Release
#
# ADD the flag, do not replace the Release flags with it. CMake composes the
# command line as ${CMAKE_C_FLAGS} ${CMAKE_C_FLAGS_RELEASE}, so CMAKE_C_FLAGS
# is the safe place: the Release defaults (/MD /O2 /Ob2 /DNDEBUG) stay as they
# are. Overwriting CMAKE_C_FLAGS_RELEASE and forgetting /MD switches the C
# runtime to the static one, and then opj_bench fails to link against it with
# LNK2038 - two halves of one program disagreeing about which CRT they use.
#
# Nothing is needed from the LINKER. /arch is a compile-time switch and leaves
# no trace in the link line. The only linker-visible requirement is the one
# above: both halves on the same C runtime, which is what not touching /MD
# gives you.
#
# Do NOT add /fp:fast to make it look better. It changes floating-point
# semantics, and the irreversible 9/7 path is floating point: the compressed
# file would come out different, and then the two builds would no longer be
# doing the same work. The comparison is only worth something while both
# builds produce the same bytes.
#
# Step 2. Build opj_bench twice, against each of them, with a tag:
#
#   cmake -S . -B build      -DOPJ_ROOT=<openjpeg> ^
#         -DOPJ_BUILD=<openjpeg>/bin       -DOPJ_BENCH_TAG=plain
#   cmake --build build --config Release
#
#   cmake -S . -B build-avx2 -DOPJ_ROOT=<openjpeg> ^
#         -DOPJ_BUILD=<openjpeg>/bin-avx2  -DOPJ_BENCH_TAG=avx2
#   cmake --build build-avx2 --config Release
#
# The two executables MUST stay in their own directories. On Windows the
# program loads openjp2.dll from beside itself, so two executables in one
# directory would both measure whichever dll happens to be there - and the
# whole comparison would quietly measure nothing. Each build directory gets
# its own copy of its own dll (see the POST_BUILD step at the bottom), so
# leaving them where CMake puts them is exactly right.
#
# The executable prints its tag at startup, and the harness records it with
# every measurement. If the two runs report the same tag, something went
# wrong and the numbers are not an A/B.

cmake_minimum_required(VERSION 3.15)
project(opj_bench CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(OPJ_ROOT "D:/_Test/openjpeg-master" CACHE PATH
    "OpenJPEG source tree")
set(OPJ_BUILD "${OPJ_ROOT}/bin" CACHE PATH
    "OpenJPEG build tree (or install prefix)")
set(OPJ_BENCH_TAG "plain" CACHE STRING "Which build this is: plain or avx2")

# --- openjpeg.h ---------------------------------------------------------
find_path(OPJ_INCLUDE_PUBLIC openjpeg.h
    HINTS "${OPJ_BUILD}/include"
          "${OPJ_BUILD}/include/openjpeg-2.5"
          "${OPJ_BUILD}/include/openjpeg-2.4"
          "${OPJ_ROOT}/src/lib/openjp2")

# --- opj_config.h, generated by the OpenJPEG build ----------------------
find_path(OPJ_INCLUDE_CONFIG opj_config.h
    HINTS "${OPJ_BUILD}/include"
          "${OPJ_BUILD}/include/openjpeg-2.5"
          "${OPJ_BUILD}/include/openjpeg-2.4"
          "${OPJ_BUILD}/src/lib/openjp2"
          "${OPJ_ROOT}/src/lib/openjp2")

# --- the import library -------------------------------------------------
find_library(OPJ_LIBRARY NAMES openjp2 libopenjp2
    HINTS "${OPJ_BUILD}/lib"
          "${OPJ_BUILD}/lib/Release"
          "${OPJ_BUILD}/bin"
          "${OPJ_BUILD}/bin/Release")

# Saying exactly which of the three is missing, and where it was looked for,
# saves a round of guessing.
if(NOT OPJ_INCLUDE_PUBLIC)
    message(FATAL_ERROR
        "openjpeg.h not found. Looked under ${OPJ_BUILD}/include and "
        "${OPJ_ROOT}/src/lib/openjp2. Pass -DOPJ_ROOT=... and, if the build "
        "tree is somewhere else, -DOPJ_BUILD=...")
endif()
if(NOT OPJ_INCLUDE_CONFIG)
    message(FATAL_ERROR
        "opj_config.h not found. It is generated by the OpenJPEG build; look "
        "for it under the build tree, usually <build>/src/lib/openjp2. Pass "
        "-DOPJ_BUILD=<that build tree>.")
endif()
if(NOT OPJ_LIBRARY)
    message(FATAL_ERROR
        "openjp2.lib not found. Looked under ${OPJ_BUILD}/lib and "
        "${OPJ_BUILD}/bin. Pass -DOPJ_BUILD=<the OpenJPEG build tree>.")
endif()

message(STATUS "OpenJPEG header:  ${OPJ_INCLUDE_PUBLIC}")
message(STATUS "OpenJPEG config:  ${OPJ_INCLUDE_CONFIG}")
message(STATUS "OpenJPEG library: ${OPJ_LIBRARY}")
message(STATUS "Build tag:        ${OPJ_BENCH_TAG}")

find_package(Threads REQUIRED)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/opj_bench-03.cpp")
    message(FATAL_ERROR
        "opj_bench-03.cpp not found next to this CMakeLists.txt. Looked in "
        "${CMAKE_CURRENT_SOURCE_DIR}")
endif()

add_executable(opj_bench opj_bench-03.cpp)
target_include_directories(opj_bench PRIVATE
    "${OPJ_INCLUDE_PUBLIC}" "${OPJ_INCLUDE_CONFIG}")
target_link_libraries(opj_bench PRIVATE "${OPJ_LIBRARY}" Threads::Threads)
# The tag is passed as a bare token and turned into a string by the
# preprocessor on the C++ side. Passing it already quoted means relying on
# CMake and MSVC agreeing about how to escape a quote inside /D, and when they
# do not the program fails to compile with an error that points at the wrong
# place entirely.
target_compile_definitions(opj_bench PRIVATE OPJ_BENCH_TAG=${OPJ_BENCH_TAG})

if(MSVC)
    target_compile_options(opj_bench PRIVATE /W3 /O2)
    target_link_libraries(opj_bench PRIVATE psapi)
else()
    target_compile_options(opj_bench PRIVATE -O2 -Wall)
endif()

# The tag says which library this executable was built against, so the two
# builds must not share an output name either: a stray copy of the wrong
# executable is harder to spot than a wrong directory.
if(NOT OPJ_BENCH_TAG STREQUAL "plain")
    set_target_properties(opj_bench PROPERTIES
        OUTPUT_NAME "opj_bench-${OPJ_BENCH_TAG}")
endif()

# On Windows the program needs openjp2.dll beside it. Copying it after the
# build removes the "works from one directory, not from another" class of
# problem entirely - and with two builds it is what keeps each executable
# talking to its own library.
if(WIN32)
    find_file(OPJ_DLL openjp2.dll
        HINTS "${OPJ_BUILD}/bin"
              "${OPJ_BUILD}/bin/Release"
              "${OPJ_BUILD}/lib"
              "${OPJ_BUILD}/lib/Release")
    if(OPJ_DLL)
        message(STATUS "OpenJPEG DLL:     ${OPJ_DLL}")
        add_custom_command(TARGET opj_bench POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${OPJ_DLL}" "$<TARGET_FILE_DIR:opj_bench>")
    else()
        message(WARNING
            "openjp2.dll not found next to the library. Copy it beside "
            "opj_bench.exe by hand, or the program will not start.")
    endif()
endif()
